Section 01

Array operation patterns

Most array operations are a loop plus one or more helper variables.

Pattern

Traversal

for (int i = 0; i < values.length; i++) {
    System.out.println(values[i]);
}

Pattern

Sum and average

int sum = 0;
for (int i = 0; i < values.length; i++) {
    sum += values[i];
}
double average = (double) sum / values.length;

Pattern

Minimum and maximum

int min = values[0];
int max = values[0];
for (int i = 1; i < values.length; i++) {
    if (values[i] < min) min = values[i];
    if (values[i] > max) max = values[i];
}

Pattern

Counting with condition

int passing = 0;
for (int score : scores) {
    if (score >= 60) {
        passing++;
    }
}

Pattern

Searching for a value

boolean found = false;
for (int i = 0; i < values.length; i++) {
    if (values[i] == target) {
        found = true;
        break;
    }
}

Full Program

Temperature analyzer

import java.util.Scanner;

public class TemperatureAnalyzer {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        double[] temperatures = new double[7];
        double sum = 0;

        for (int i = 0; i < temperatures.length; i++) {
            System.out.print("Temperature day " + (i + 1) + ": ");
            temperatures[i] = input.nextDouble();
            sum += temperatures[i];
        }

        double min = temperatures[0];
        double max = temperatures[0];
        int aboveThirty = 0;

        for (int i = 0; i < temperatures.length; i++) {
            if (temperatures[i] < min) min = temperatures[i];
            if (temperatures[i] > max) max = temperatures[i];
            if (temperatures[i] >= 30) aboveThirty++;
        }

        System.out.printf("Average: %.2f%n", sum / temperatures.length);
        System.out.println("Minimum: " + min);
        System.out.println("Maximum: " + max);
        System.out.println("Days >= 30: " + aboveThirty);
        input.close();
    }
}

Copy

Copying values

int[] copy = new int[values.length];
for (int i = 0; i < values.length; i++) {
    copy[i] = values[i];
}

Reverse

Reverse print without changing array

for (int i = values.length - 1; i >= 0; i--) {
    System.out.println(values[i]);
}

Debug

Wrong initial max

int max = 0; // risky if all values are negative
for (int value : values) {
    if (value > max) max = value;
}

Safer: initialize from the first array element.
