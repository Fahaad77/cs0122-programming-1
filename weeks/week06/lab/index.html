Interactive Repeating Menu

Build a menu that repeats until Exit, uses switch, validates choices, prints nested-loop patterns, and demonstrates break/continue.

Objective

Build a menu that repeats until Exit, uses switch, validates choices, prints nested-loop patterns, and demonstrates break/continue.

  • Use do-while when the body must run at least once.
  • Trace nested loops with rows and columns.
  • Use break to leave a loop and continue to skip an iteration.
  • Explain return as leaving the current method without turning CS0122 into an OOP course.

Starter code

import java.util.Scanner;

public class RepeatingMenu {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int choice;

        do {
            System.out.println("1. Print table");
            System.out.println("2. Find first divisor");
            System.out.println("3. Skip negatives in sample data");
            System.out.println("0. Exit");
            System.out.print("Choice: ");
            choice = input.nextInt();

            switch (choice) {
                case 1:
                    for (int row = 1; row <= 5; row++) {
                        for (int col = 1; col <= 5; col++) {
                            System.out.printf("%4d", row * col);
                        }
                        System.out.println();
                    }
                    break;
                case 2:
                    System.out.print("Number: ");
                    int number = input.nextInt();
                    for (int d = 2; d <= number; d++) {
                        if (number % d == 0) {
                            System.out.println("First divisor: " + d);
                            break;
                        }
                    }
                    break;
                case 3:
                    int[] values = {4, -2, 7, -9, 3};
                    for (int value : values) {
                        if (value < 0) {
                            continue;
                        }
                        System.out.println("Accepted: " + value);
                    }
                    break;
                case 0:
                    System.out.println("Goodbye");
                    break;
                default:
                    System.out.println("Invalid choice");
            }
        } while (choice != 0);

        input.close();
    }
}
